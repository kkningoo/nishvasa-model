%% Model 3 — Bead Effectiveness Factor
% Checks whether intraparticle diffusion limits the immobilized enzyme at
% the corrected bead radius (10 um, constrained by coating geometry, not
% by diffusion). Compares sphere/cylinder/slab shapes.

clear; clc; close all;

% --- Physical parameters ---
A_GS = 0.0783333;             % mol s^-1 kg^-1 (4.7 umol/min/mg)
M_LOAD = 2.0;                 % kg/m^3
A_M = A_GS * M_LOAD;          % mol s^-1 m^-3

D_FREE_ATP = 3.7e-10;         % m^2/s
D_FREE_NH3 = 1.6e-9;
D_FREE_GLU = 7.6e-10;
GEL_RETARDATION = 0.5;

D_ATP_GEL = GEL_RETARDATION * D_FREE_ATP;   % 1.85e-10
D_NH3_GEL = GEL_RETARDATION * D_FREE_NH3;   % 8.0e-10
D_GLU_GEL = GEL_RETARDATION * D_FREE_GLU;   % 3.8e-10

substrates = {
    'ATP',       7.0 + 3.5,   D_ATP_GEL;
    'NH3',       0.4 + 2.95,  8.2e-10;
    'Glutamate', 30.0 + 150.0, 3.8e-10
};

L_COAT = 24e-6;               % m
L_BEAD_SELECTED = 10e-6;      % m

% --- Print Thiele modulus for each substrate at L = 10 um ---
fprintf('Thiele modulus at L = 10 um, M = 2.0 kg/m^3:\n');
for i = 1:size(substrates, 1)
    sub_name = substrates{i, 1};
    kmc = substrates{i, 2};
    D_val = substrates{i, 3};
    phi = thiele_modulus(L_BEAD_SELECTED, A_M, kmc, D_val);
    eta = eta_sphere(phi);
    fprintf('  %-12s: Km+c = %6.2f mM, D = %.2e m^2/s, phi = %.4f, eta_sphere = %.4f\n', ...
        sub_name, kmc, D_val, phi, eta);
end

% --- Diffusivity audit ---
D_implied_ATP = 4.8e-12;
D_implied_NH3 = 1.5e-11;
fprintf('\nDiffusivity audit (withdrawn tuned constant):\n');
fprintf('  Implied D_ATP = %.1e m^2/s  vs  free = %.1e (%.0fx higher)\n', ...
    D_implied_ATP, D_FREE_ATP, D_FREE_ATP / D_implied_ATP);
fprintf('  Implied D_NH3 = %.1e m^2/s  vs  free = %.1e (%.0fx higher)\n', ...
    D_implied_NH3, D_FREE_NH3, D_FREE_NH3 / D_implied_NH3);
fprintf('  Both are 40-100x below the plausible gel range. Tuned constant discarded.\n');

% --- Bead radius constraint ---
L_max = L_COAT / 2;
fprintf('\nBead radius constraint: L_bead <= L_coat/2 = %.0f um\n', L_max * 1e6);
fprintf('Selected: L_bead = %.0f um (leaves margin for coating variation)\n', L_BEAD_SELECTED * 1e6);
fprintf('Leaching caveat: 10 um is below the 34-500 um chitosan microsphere range. Enzyme retention at this size is a wet-lab question, not captured by eta.\n');

% --- Shape comparison at phi = 5 ---
phi_check = 5.0;
fprintf('\nShape comparison at phi = %.1f:\n', phi_check);
fprintf('  eta_sphere   = %.2f\n', eta_sphere(phi_check));
fprintf('  eta_cylinder = %.2f\n', eta_cylinder(phi_check));
fprintf('  eta_slab     = %.2f\n', eta_slab(phi_check));

% eta at M = 30 kg/m^3, L = 10 um
phi_30 = thiele_modulus(L_BEAD_SELECTED, A_GS * 30.0, 10.5, D_ATP_GEL);
eta_30 = eta_sphere(phi_30);
fprintf('\nAt M = 30 kg/m^3, L = 10 um: phi = %.3f, eta_sphere = %.4f\n', phi_30, eta_30);

% ============================================================
% Figure 1: eta_bead vs assumed ATP diffusivity
% ============================================================
D_sweep = logspace(-12, -9, 500);
phi_D = thiele_modulus(L_BEAD_SELECTED, A_M, 10.5, D_sweep);
eta_D = eta_sphere(phi_D);

fig1 = figure('Units', 'inches', 'Position', [1, 1, 8, 5], 'Visible', 'off');
ax1 = axes(fig1); hold(ax1, 'on'); box(ax1, 'on');
set(ax1, 'XScale', 'log');

D_gel_lo = 0.3 * D_FREE_ATP;
D_gel_hi = 0.8 * D_FREE_ATP;
fill(ax1, [D_gel_lo, D_gel_hi, D_gel_hi, D_gel_lo], [0.9, 0.9, 1.005, 1.005], ...
    [0, 158, 115]/255, 'FaceAlpha', 0.15, 'EdgeColor', 'none', ...
    'DisplayName', 'Plausible gel range (0.3-0.8 x free)');

plot(ax1, D_sweep, eta_D, 'Color', [0, 114, 178]/255, 'LineWidth', 2, 'HandleVisibility', 'off');
xline(ax1, D_ATP_GEL, '--', 'Color', [0, 158, 115]/255, 'LineWidth', 1.2, ...
    'DisplayName', sprintf('Stated D = %.2e', D_ATP_GEL));
xline(ax1, D_implied_ATP, ':', 'Color', [213, 94, 0]/255, 'LineWidth', 1.5, ...
    'DisplayName', sprintf('Withdrawn (tuned) D = %.1e', D_implied_ATP));

xlabel(ax1, 'Assumed $D_{\mathrm{ATP,gel}}$ (m$^2$/s)', 'Interpreter', 'latex', 'FontSize', 11);
ylabel(ax1, '$\eta_{\mathrm{bead}}$', 'Interpreter', 'latex', 'FontSize', 11);
title(ax1, {'Bead effectiveness factor vs assumed ATP diffusivity', ...
    '($L$ = 10 $\mu$m, $M$ = 2.0 kg/m$^3$)'}, 'Interpreter', 'latex', 'FontSize', 12);
legend(ax1, 'Location', 'southeast', 'FontSize', 9);
xlim(ax1, [1e-12, 1e-9]); ylim(ax1, [0.9, 1.005]);
exportgraphics(fig1, 'model3_fig1_eta_vs_diffusivity.png', 'Resolution', 200);
close(fig1);

% ============================================================
% Figure 2: eta vs Thiele modulus for three shapes
% ============================================================
phi_range = logspace(-2, 1.5, 500);
fig2 = figure('Units', 'inches', 'Position', [1, 1, 8, 5], 'Visible', 'off');
ax2 = axes(fig2); hold(ax2, 'on'); box(ax2, 'on');
set(ax2, 'XScale', 'log');

plot(ax2, phi_range, eta_sphere(phi_range), 'Color', [0, 114, 178]/255, 'LineWidth', 2, 'DisplayName', 'Sphere');
plot(ax2, phi_range, eta_cylinder(phi_range), 'Color', [230, 159, 0]/255, 'LineWidth', 2, 'DisplayName', 'Cylinder');
plot(ax2, phi_range, eta_slab(phi_range), 'Color', [213, 94, 0]/255, 'LineWidth', 2, 'DisplayName', 'Slab');

phi_op = thiele_modulus(L_BEAD_SELECTED, A_M, 10.5, D_ATP_GEL);
xline(ax2, phi_op, '--', 'Color', [0.5, 0.5, 0.5], 'HandleVisibility', 'off');
text(ax2, phi_op * 1.5, 0.80, sprintf('Operating \\phi = %.3f', phi_op), 'Color', [0.4, 0.4, 0.4], 'FontSize', 9);

xlabel(ax2, 'Thiele modulus $\phi$', 'Interpreter', 'latex', 'FontSize', 11);
ylabel(ax2, '$\eta$', 'Interpreter', 'latex', 'FontSize', 11);
title(ax2, 'Effectiveness factor vs Thiele modulus (Aris 1975)', 'FontSize', 12);
legend(ax2, 'Location', 'southwest', 'FontSize', 10);
xlim(ax2, [1e-2, 10^1.5]); ylim(ax2, [0, 1.05]);
exportgraphics(fig2, 'model3_fig2_eta_vs_phi_shapes.png', 'Resolution', 200);
close(fig2);

% ============================================================
% Figure 3: eta map over bead radius and loading
% ============================================================
L_range = logspace(log10(5e-6), log10(200e-6), 200);
M_range = logspace(log10(1.0), log10(30.0), 200);
[L_grid, M_grid] = meshgrid(L_range, M_range);
phi_grid = thiele_modulus(L_grid, A_GS .* M_grid, 10.5, D_ATP_GEL);
eta_grid = eta_sphere(phi_grid);

fig3 = figure('Units', 'inches', 'Position', [1, 1, 8, 6], 'Visible', 'off');
ax3 = axes(fig3); hold(ax3, 'on'); box(ax3, 'on');
pcolor(ax3, L_range * 1e6, M_range, eta_grid);
shading(ax3, 'flat');
set(ax3, 'XScale', 'log', 'YScale', 'log');
clim(ax3, [0.9, 1.0]);
colormap(ax3, parula);
cb = colorbar(ax3);
cb.Label.String = '\eta_{sphere}';

xline(ax3, 12.0, '--w', 'LineWidth', 1.5);
text(ax3, 12.5, 25, sprintf('L <= 12 um\n(coating fit)'), 'Color', 'w', 'FontSize', 8, 'VerticalAlignment', 'top');
plot(ax3, 10, 2.0, 'p', 'MarkerFaceColor', 'w', 'MarkerEdgeColor', 'k', 'MarkerSize', 12);
text(ax3, 13, 2.0, sprintf('Selected\n(10 um, 2 kg/m^3)'), 'Color', 'w', 'FontSize', 8);

xlabel(ax3, 'Bead radius (\mum)', 'FontSize', 11);
ylabel(ax3, 'Enzyme loading $M$ (kg/m$^3$)', 'Interpreter', 'latex', 'FontSize', 11);
title(ax3, '$\eta_{\mathrm{sphere}}$ over bead radius and loading (ATP-limited)', 'Interpreter', 'latex', 'FontSize', 12);
xlim(ax3, [5, 200]); ylim(ax3, [1, 30]);
exportgraphics(fig3, 'model3_fig3_eta_map.png', 'Resolution', 200);
close(fig3);

fprintf('\nConclusion: At L = 10 um, M = 2.0 kg/m^3, eta_bead = 0.9995. Diffusion does not limit the bead. Bead size is set by coating geometry (L <= 12 um), not diffusion.\n');

% --- Helper Functions ---
function phi = thiele_modulus(L, a_M, km_plus_c, D)
    phi = L .* sqrt(a_M ./ (km_plus_c .* D));
end

function eta = eta_sphere(phi)
    eta = ones(size(phi));
    large = phi > 1e-2;
    p_l = phi(large);
    eta(large) = (3.0 ./ p_l) .* (1.0 ./ tanh(p_l) - 1.0 ./ p_l);
    small = ~large;
    p_s = phi(small);
    eta(small) = 1.0 - (p_s.^2) ./ 15.0;
end

function eta = eta_cylinder(phi)
    eta = ones(size(phi));
    large = phi > 1e-6;
    p_l = phi(large);
    eta(large) = (2.0 ./ p_l) .* besseli(1, p_l) ./ besseli(0, p_l);
end

function eta = eta_slab(phi)
    eta = ones(size(phi));
    large = phi > 1e-6;
    p_l = phi(large);
    eta(large) = tanh(p_l) ./ p_l;
end
