%% Model 6 — Single-Pass Removal Rate of One Cartridge
% Combines enzyme kinetics with plug-flow integration along the channel
% to give the actual removal rate.

clear; clc; close all;

% --- Physical constants ---
R_GAS = 8.314462;
T_K = 298.15;
P_ATM = 101325.0;
p.PPM_TO_MOLM3 = P_ATM / (R_GAS * T_K) * 1e-6;

p.H_CC = 1443.0;
p.D_NH3_AIR = 2.3e-5;
p.SH = 3.0;
p.MU_AIR = 1.84e-5;
p.RHO_AIR = 1.184;

% --- Scaffold geometry ---
p.L_COAT = 24e-6;
R_CELL = 1.5e-3;
T_WALL = 0.8e-3;
p.PACK_FRACTION = 0.92;
Z_TRAY = 0.020;
N_TRAYS = 14;
p.Z_STACK = N_TRAYS * Z_TRAY;

p.r_open = R_CELL - p.L_COAT;
p.r_out = R_CELL + T_WALL / 2;
p.A_open = 2 * sqrt(3) * p.r_open^2;
p.A_cell = 2 * sqrt(3) * p.r_out^2;
p.P_wet = 4 * sqrt(3) * p.r_open;
p.D_h = 2 * p.r_open;
p.eps_open = p.A_open / p.A_cell;

% --- Kinetic parameters ---
A_GS = 0.0783333;
M_LOAD = 2.0;
ETA_BEAD = 1.00;
ETA_COAT = 0.985;
RHO_RETAINED = 0.5;

KM_GLU_MM = 30.0;
KM_ATP_MM = 7.0;
p.KM_NH3_MOLM3 = 0.4;
GLU_FIXED_MM = 150.0;
ATP_FIXED_MM = 3.5;

f_Glu = GLU_FIXED_MM / (KM_GLU_MM + GLU_FIXED_MM);
f_ATP = ATP_FIXED_MM / (KM_ATP_MM + ATP_FIXED_MM);

vmax_eff_fn = @(rho) ETA_BEAD * ETA_COAT * rho * A_GS * M_LOAD * f_Glu * f_ATP;
V_MAX_EFF = vmax_eff_fn(RHO_RETAINED);
V_MAX_EFF_LO = vmax_eff_fn(0.3);
V_MAX_EFF_HI = vmax_eff_fn(0.7);

p.k_g = p.SH * p.D_NH3_AIR / p.D_h;

% ============================================================
% Print expected results: D = 400 mm, u_face = 0.4 m/s, 50 ppm
% ============================================================
m = cartridge_metrics(50.0, 0.4, 400, V_MAX_EFF, p);
fprintf('=== D = 400 mm, u_face = 0.4 m/s, 50 ppm ===\n');
fprintf('N_cells       = %d\n', m.N_cells);
fprintf('TCWA          = %.2f m^2\n', m.TCWA);
fprintf('Q             = %.1f m^3/h\n', m.Q_m3h);
fprintf('u_ch          = %.2f m/s\n', m.u_ch);
fprintf('Re            = %.0f (laminar)\n', m.Re);
fprintf('k_g           = %.4f m/s\n', p.k_g);
fprintf('R             = %.0f umol/min\n', m.R_umol_min);
fprintf('R_kin         = %.0f umol/min\n', m.R_kin_umol_min);
fprintf('R_del         = %.0f umol/min  (%.1fx removal)\n', m.R_del_umol_min, m.R_del_umol_min / m.R_umol_min);
fprintf('Single-pass X = %.1f%%\n', m.X * 100);
fprintf('C_out         = %.1f ppm\n', m.C_out_ppm);
fprintf('Film loss     = %.2f%%\n', m.film_loss_pct);
fprintf('dP scaffold   = %.1f Pa\n', m.dP);
fprintf('The unit is kinetics-limited, not supply-limited.\n');

fprintf('\nNote: a nominal 16" PVC pipe has OD ~406 mm; inner diameter is below 400 mm. A 385 mm disc would have ~%.0f%% fewer cells.\n', ...
    (1 - (385/400)^2)*100);
fprintf('\nPressure budget: scaffold %.1f Pa + pre-filter 20-50 Pa + humidifier 30-100 Pa = %.0f-%.0f Pa total\n', ...
    m.dP, m.dP + 20, m.dP + 150);

u_sweep_cross = linspace(0.01, 0.15, 1000);
for u = u_sweep_cross
    mc = cartridge_metrics(50.0, u, 400, V_MAX_EFF, p);
    if mc.R_del_umol_min >= mc.R_kin_umol_min
        fprintf('Delivery = kinetic ceiling at u_face ~ %.3f m/s\n', u);
        break;
    end
end

for u_test = [0.3, 0.5]
    mc = cartridge_metrics(50.0, u_test, 400, V_MAX_EFF, p);
    gap = (1.0 - mc.R_umol_min / mc.R_kin_umol_min) * 100;
    fprintf('At u_face = %.1f: R = %.0f, gap to R_kin = %.1f%%\n', u_test, mc.R_umol_min, gap);
end

fprintf('\nConcentration band (rho = 0.5 nominal, 0.3/0.7 band):\n');
for c_ppm = [20, 50, 100]
    c_liq = p.H_CC * c_ppm * p.PPM_TO_MOLM3;
    f_nh3 = c_liq / (p.KM_NH3_MOLM3 + c_liq);
    r_nom = cartridge_metrics(c_ppm, 0.4, 400, V_MAX_EFF, p).R_umol_min;
    r_lo = cartridge_metrics(c_ppm, 0.4, 400, V_MAX_EFF_LO, p).R_umol_min;
    r_hi = cartridge_metrics(c_ppm, 0.4, 400, V_MAX_EFF_HI, p).R_umol_min;
    fprintf('  %3d ppm: R = %.0f (%.0f-%.0f) umol/min, f_NH3 = %.1f%%\n', ...
        c_ppm, r_nom, r_lo, r_hi, f_nh3 * 100);
end

fprintf('\nDiameter table (u_face = 0.4 m/s, rho = 0.5):\n');
fprintf('%8s %7s %10s %10s %10s %10s\n', 'D (mm)', 'Cells', 'TCWA (m^2)', 'R@50ppm', 'Q (m^3/h)', 'PETG (kg)');
for D_mm = [200, 250, 300, 400, 500]
    mc = cartridge_metrics(50.0, 0.4, D_mm, V_MAX_EFF, p);
    wall_vol = mc.N_cells * (p.A_cell - p.A_open) * p.Z_STACK;
    petg_kg = wall_vol * 1270;
    fprintf('%8d %7d %10.2f %10.0f %10.1f %10.1f\n', ...
        D_mm, mc.N_cells, mc.TCWA, mc.R_umol_min, mc.Q_m3h, petg_kg);
end

% --- Helper Functions ---
function [j, C_s] = wall_flux(C_gas, v_max_eff_local, p)
    jm = v_max_eff_local * p.L_COAT;
    a_coeff = p.k_g * p.H_CC;
    b_coeff = -(p.k_g * p.H_CC * C_gas - p.k_g * p.KM_NH3_MOLM3 - jm * p.H_CC);
    c_coeff = -p.k_g * p.KM_NH3_MOLM3 * C_gas;
    
    disc = max(b_coeff^2 - 4 * a_coeff * c_coeff, 0);
    C_s = (-b_coeff + sqrt(disc)) / (2 * a_coeff);
    c_liq = p.H_CC * C_s;
    f_nh3 = c_liq / (p.KM_NH3_MOLM3 + c_liq);
    j = v_max_eff_local * p.L_COAT * f_nh3;
end

function [C_out, u_ch] = plug_flow_outlet(C_in, u_face, v_max_eff_local, p)
    u_ch = u_face / (p.PACK_FRACTION * p.eps_open);
    opts = odeset('RelTol', 1e-10, 'AbsTol', 1e-14, 'MaxStep', p.Z_STACK / 100);
    [~, C_sol] = ode45(@(x, C) dC_dx(x, C, u_ch, v_max_eff_local, p), [0, p.Z_STACK], C_in, opts);
    C_out = max(C_sol(end), 0.0);
end

function dC = dC_dx(~, C, u_ch, v_max_eff_local, p)
    if C <= 0
        dC = 0.0;
        return;
    end
    [j, ~] = wall_flux(C, v_max_eff_local, p);
    dC = -(4.0 / (p.D_h * u_ch)) * j;
end

function m = cartridge_metrics(C_in_ppm, u_face, D_cart_mm, v_max_eff_local, p)
    C_in = C_in_ppm * p.PPM_TO_MOLM3;
    D_cart = D_cart_mm * 1e-3;
    m.N_cells = floor(p.PACK_FRACTION * pi * D_cart^2 / 4 / p.A_cell);
    m.Q_m3s = u_face * pi * D_cart^2 / 4;
    m.Q_m3h = m.Q_m3s * 3600;
    m.TCWA = m.N_cells * p.P_wet * p.Z_STACK;
    
    [C_out, m.u_ch] = plug_flow_outlet(C_in, u_face, v_max_eff_local, p);
    R_actual = m.Q_m3s * (C_in - C_out);
    R_del = m.Q_m3s * C_in;
    [j_in, ~] = wall_flux(C_in, v_max_eff_local, p);
    R_kin = j_in * m.TCWA;
    
    m.C_out_ppm = C_out / p.PPM_TO_MOLM3;
    m.R_umol_min = R_actual * 1e6 * 60;
    m.R_del_umol_min = R_del * 1e6 * 60;
    m.R_kin_umol_min = R_kin * 1e6 * 60;
    m.X = 1.0 - C_out / C_in;
    m.Re = p.RHO_AIR * m.u_ch * p.D_h / p.MU_AIR;
    m.dP = 32 * p.MU_AIR * m.u_ch * p.Z_STACK / p.D_h^2;
    
    c_liq_direct = p.H_CC * C_in;
    f_nh3_direct = c_liq_direct / (p.KM_NH3_MOLM3 + c_liq_direct);
    j_direct = v_max_eff_local * p.L_COAT * f_nh3_direct;
    m.film_loss_pct = (1.0 - j_in / j_direct) * 100;
end
