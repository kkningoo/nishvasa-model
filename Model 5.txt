%% Model 5 — Honeycomb Scaffold Cell Geometry
% Picks the hexagonal cell inradius r by maximising VRE (volumetric
% removal efficiency = GSA * eps_open).

clear; clc; close all;

% --- Parameters ---
L_COAT = 24e-6;     % m
T_WALL = 0.8e-3;    % m

% Analytical VRE optimum
R_STAR = 4 * L_COAT + 1.5 * T_WALL;
fprintf('r* = 4*L_coat + 1.5*t_w = %.3f mm\n', R_STAR * 1e3);

r_open_star = R_STAR - L_COAT;
r_out_star = R_STAR + T_WALL / 2;
ratio_check = r_open_star / r_out_star;
eps_check = ratio_check^2;
fprintf('At r*: r_open/r_out = %.4f (expect 0.75), eps_open = %.4f (expect 0.5625 = 9/16)\n', ...
    ratio_check, eps_check);

% --- Sweep r from 0.3 mm to 5 mm ---
r_sweep = linspace(0.3e-3, 5e-3, 2000);
r_mm = r_sweep * 1e3;

vre_arr = zeros(size(r_sweep));
for i = 1:length(r_sweep)
    g = cell_geometry(r_sweep(i), L_COAT, T_WALL);
    vre_arr(i) = g.VRE;
end

[vre_peak, i_vrepeak] = max(vre_arr);
vre_norm = vre_arr / vre_peak;
r_vrepeak_mm = r_mm(i_vrepeak);

above_97 = r_mm(vre_norm >= 0.97);
plateau_lo = above_97(1);
plateau_hi = above_97(end);

% --- Design Score (illustrative heuristic) ---
R_ACCESS = 0.45e-3;
D_MIN = 0.8e-3;

F_coat = 1.0 - exp(-r_sweep / R_ACCESS);
F_fdm = 1.0 - (D_MIN ./ (2 * r_sweep)).^2;
design_score = vre_arr .* F_coat .* F_fdm;
[ds_peak, i_dspeak] = max(design_score);
ds_norm = design_score / ds_peak;
r_dspeak_mm = r_mm(i_dspeak);

% --- Print table ---
fprintf('\nVRE peak at r = %.3f mm (= r*)\n', r_vrepeak_mm);
fprintf('Design Score peak at r = %.2f mm\n', r_dspeak_mm);
fprintf('97%% VRE plateau: %.1f to %.1f mm\n\n', plateau_lo, plateau_hi);

test_radii_mm = [0.5, 1.0, 1.5, 2.0, 3.0];
fprintf('%8s  %10s  %8s  %10s  %10s\n', 'r (mm)', 'eps_open (%)', 'D_h (mm)', 'VRE (rel)', 'DS (rel)');
for rm = test_radii_mm
    g = cell_geometry(rm * 1e-3, L_COAT, T_WALL);
    vn = g.VRE / vre_peak;
    dsn = (g.VRE * (1.0 - exp(-rm*1e-3 / R_ACCESS)) * (1.0 - (D_MIN / (2 * rm * 1e-3))^2)) / ds_peak;
    fprintf('%8.1f  %10.1f  %8.2f  %10.2f  %10.2f\n', rm, g.eps_open*100, g.D_h*1e3, vn, dsn);
end

g_sel = cell_geometry(1.5e-3, L_COAT, T_WALL);
vn_sel = g_sel.VRE / vre_peak;
fprintf('\nSelected r = 1.5 mm retains %.1f%% of VRE peak.\n', vn_sel * 100);
fprintf('Geometry: r_open = %.3f mm, D_h = %.2f mm, A_open = %.2f mm^2, A_cell = %.1f mm^2, P = %.1f mm, eps_open = %.1f%%\n', ...
    g_sel.r_open*1e3, g_sel.D_h*1e3, g_sel.A_open*1e6, g_sel.A_cell*1e6, g_sel.P*1e3, g_sel.eps_open*100);

hex_to_circle = 2 * pi / (3 * sqrt(3));
fprintf('\nCircumscribed-circle / hex area ratio = %.3f (a %.0f%% overestimate)\n', ...
    hex_to_circle, (hex_to_circle - 1)*100);

% ============================================================
% Figure: normalised VRE and Design Score vs r
% ============================================================
fig = figure('Units', 'inches', 'Position', [1, 1, 9, 5.5], 'Visible', 'off');
ax = axes(fig); hold(ax, 'on'); box(ax, 'on');

fill(ax, [0.3, 5.0, 5.0, 0.3], [0.97, 0.97, 1.0, 1.0], [0, 114, 178]/255, ...
    'FaceAlpha', 0.08, 'EdgeColor', 'none', 'HandleVisibility', 'off');
yline(ax, 0.97, ':', 'Color', [0, 114, 178]/255, 'HandleVisibility', 'off');
text(ax, 4.8, 0.985, '>= 97% of VRE peak', 'Color', [0, 114, 178]/255, ...
    'FontSize', 8, 'HorizontalAlignment', 'right');

plot(ax, r_mm, vre_norm, '-', 'Color', [0, 114, 178]/255, 'LineWidth', 2.5, 'DisplayName', 'VRE (normalised)');
plot(ax, r_mm, ds_norm, '--', 'Color', [230, 159, 0]/255, 'LineWidth', 2, 'DisplayName', 'Design Score (illustrative)');

xline(ax, r_vrepeak_mm, ':', 'Color', [0, 114, 178]/255, 'HandleVisibility', 'off');
text(ax, r_vrepeak_mm + 0.3, 0.90, sprintf('VRE peak\nr* = %.2f mm', r_vrepeak_mm), ...
    'Color', [0, 114, 178]/255, 'FontSize', 9);

plot(ax, 1.5, vn_sel, 'o', 'Color', [0, 114, 178]/255, 'MarkerFaceColor', [0, 114, 178]/255, ...
    'MarkerSize', 8, 'HandleVisibility', 'off');
text(ax, 2.1, 0.82, sprintf('Selected\nr = 1.5 mm\n(%.1f%% of peak)', vn_sel*100), ...
    'Color', [0, 114, 178]/255, 'FontSize', 9);

text(ax, r_dspeak_mm + 0.4, 0.72, sprintf('DS peak\n%.2f mm', r_dspeak_mm), ...
    'Color', [230, 159, 0]/255, 'FontSize', 9);

xlabel(ax, 'Cell inradius r (mm)', 'FontSize', 11);
ylabel(ax, 'Normalised value', 'FontSize', 11);
title(ax, {'VRE and Design Score vs cell inradius', ...
    sprintf('(L_{coat} = %.0f um, t_w = %.1f mm)', L_COAT*1e6, T_WALL*1e3)}, 'FontSize', 12);
legend(ax, 'Location', 'southeast', 'FontSize', 10);
xlim(ax, [0.3, 5.0]); ylim(ax, [0, 1.08]);

exportgraphics(fig, 'model5_scaffold_geometry.png', 'Resolution', 200);
close(fig);

fprintf('\nConclusion: r = 1.5 mm is inside the >= 97%% VRE plateau (1.0-1.7 mm). It is selected for its higher open fraction and printability, not at either peak.\n');

% --- Helper Functions ---
function g = cell_geometry(r, L_COAT, T_WALL)
    g.r_open = r - L_COAT;
    g.r_out = r + T_WALL / 2;
    g.A_open = 2 * sqrt(3) * g.r_open^2;
    g.A_cell = 2 * sqrt(3) * g.r_out^2;
    g.P = 4 * sqrt(3) * g.r_open;
    g.D_h = 2 * g.r_open;
    g.eps_open = g.A_open / g.A_cell;
    g.GSA = g.P / g.A_cell;
    g.VRE = g.GSA * g.eps_open;
end
