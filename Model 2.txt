%% Model 2 — ATP Regeneration and Reagent Life
% Model 2A: coating liquid alone, showing PPK2 necessity.
% Model 2B: reservoir-coupled 7-day run for 10/30/100 L reservoirs.
% All concentrations in mM. Time in minutes (2A) or hours/days (2B).

clear; clc; close all;

% --- GS parameters (H195Y, Fisher & Wray 2006) ---
params.KM_GLU_MM = 30.0;
params.KM_ATP_MM = 7.0;
params.A_GS_UMOL_MIN_MG = 4.7;       % specific activity
params.KM_NH3_MM = 0.4;              % Nocek et al. (2008)
params.KI_GLN_MM = 19.167;           % Cheng–Prusoff derived (Model 1)

% --- PPK2 parameters (SlPPK2, Sun et al. 2021, Table 2) ---
params.A_PPK2_UMOL_MIN_MG = 283.0;
params.KM_ADP_MM = 0.61;
params.KM_POLYP_MM = 1.96;

% --- Device loading ---
M_KG_M3 = 2.0;
ETA_BEAD = 1.00;                     % Model 3 corrected result at 10 um
ETA_COAT = 0.985;                    % Model 4
RHO_RETAINED = 0.5;
E_ACTIVE_MG_ML = ETA_BEAD * ETA_COAT * RHO_RETAINED * M_KG_M3; % kg/m^3 == mg/mL

params.VMAX_GS = params.A_GS_UMOL_MIN_MG * E_ACTIVE_MG_ML;     % mM/min
params.VMAX_PPK2 = params.A_PPK2_UMOL_MIN_MG * E_ACTIVE_MG_ML;

% --- Reservoir charge ---
params.GLU_0_MM = 150.0;
params.ATP_0_MM = 3.5;
params.ADP_0_MM = 0.0;
params.POLYP_0_MM = 30.0;
params.GLN_0_MM = 0.0;
params.PI_0_MM = 0.0;

% --- Model 2A constants ---
params.NH3_DISSOLVED_MM = 2.95;      % equilibrium with 50 ppm via Henry's law
PHOSPHATE_INHIB_THRESHOLD_MM = 12.5; % BNU-China 2025

% ============================================================
% Model 2A — Coating liquid alone, fixed dissolved NH3
% ============================================================
y0_2a = [params.GLU_0_MM; params.GLN_0_MM; params.PI_0_MM; ...
         params.ATP_0_MM; params.ADP_0_MM; params.POLYP_0_MM];
t_eval_2a = linspace(0, 60, 2000);
opts = odeset('RelTol', 1e-8, 'AbsTol', 1e-10);

[t_no, y_no] = ode15s(@(t, y) ode_2a(t, y, false, params), t_eval_2a, y0_2a, opts);
[t_with, y_with] = ode15s(@(t, y) ode_2a(t, y, true, params), t_eval_2a, y0_2a, opts);

% When does ATP drop below 0.5 mM without PPK2?
atp_no = y_no(:, 4);
idx_05 = find(atp_no < 0.5, 1, 'first');
if ~isempty(idx_05)
    fprintf('Model 2A: Without PPK2, ATP < 0.5 mM at t = %.1f min\n', t_no(idx_05));
end

% ATP with PPK2
atp_with = y_with(:, 4);
fprintf('Model 2A: With PPK2, ATP minimum = %.2f mM\n', min(atp_with));

% PolyP exhaustion time
polyp_with = y_with(:, 6);
idx_polyp = find(polyp_with < 0.01, 1, 'first');
if ~isempty(idx_polyp)
    fprintf('Model 2A: PolyP exhausted at t = %.0f min\n', t_with(idx_polyp));
end

% --- Figure 2A: with/without PPK2 comparison ---
fig_2a = figure('Units', 'inches', 'Position', [1, 1, 14, 5], 'Visible', 'off');
tl = tiledlayout(fig_2a, 1, 2, 'TileSpacing', 'compact', 'Padding', 'compact');

ax_a = nexttile(tl); hold(ax_a, 'on'); box(ax_a, 'on');
plot(ax_a, t_no, y_no(:, 4), 'Color', [213, 94, 0]/255, 'LineWidth', 2, 'DisplayName', 'Without PPK2');
plot(ax_a, t_with, y_with(:, 4), 'Color', [0, 114, 178]/255, 'LineWidth', 2, 'DisplayName', 'With PPK2');
xlabel(ax_a, 'Time (min)', 'FontSize', 11);
ylabel(ax_a, '[ATP] (mM)', 'FontSize', 11);
title(ax_a, '(a) ATP pool', 'FontSize', 12);
legend(ax_a, 'Location', 'best', 'FontSize', 10);
xlim(ax_a, [0, 60]); ylim(ax_a, [0, inf]);

ax_b = nexttile(tl); hold(ax_b, 'on'); box(ax_b, 'on');
plot(ax_b, t_with, y_with(:, 6), 'Color', [0, 158, 115]/255, 'LineWidth', 2, 'DisplayName', 'PolyP');
plot(ax_b, t_with, y_with(:, 2), 'Color', [204, 121, 167]/255, 'LineWidth', 2, 'DisplayName', 'Glutamine');
plot(ax_b, t_with, y_with(:, 3), 'Color', [86, 180, 233]/255, 'LineWidth', 2, 'DisplayName', 'P_i');
xlabel(ax_b, 'Time (min)', 'FontSize', 11);
ylabel(ax_b, 'Concentration (mM)', 'FontSize', 11);
title(ax_b, '(b) Coupled system species', 'FontSize', 12);
legend(ax_b, 'Location', 'best', 'FontSize', 10);
xlim(ax_b, [0, 60]); ylim(ax_b, [0, inf]);

exportgraphics(fig_2a, 'model2a_ppk2_necessity.png', 'Resolution', 200);
close(fig_2a);

% ============================================================
% Model 2B — Reservoir-coupled 7-day simulation
% ============================================================
DAILY_LOAD_MOL = 0.307;
params.DAILY_LOAD_MM_PER_L = DAILY_LOAD_MOL * 1000.0; % mmol per day per unit
params.V_COAT_L_ACTUAL = 0.64;                        % litres

reservoir_sizes = [10, 30, 100];
res_colors = {[213, 94, 0]/255, [230, 159, 0]/255, [0, 114, 178]/255};

fprintf('\n--- Model 2B: 7-day reservoir simulation ---\n');

all_scrub_times = cell(1, 3);
all_compositions = cell(1, 3);

for i = 1:length(reservoir_sizes)
    vr = reservoir_sizes(i);
    [times, comps] = simulate_reservoir_7days(vr, params);
    all_scrub_times{i} = times;
    all_compositions{i} = comps;
    fprintf('\nReservoir %d L:\n', vr);
    for d = 1:7
        if ~isnan(times(d))
            fprintf('  Day %d: %.1f h\n', d, times(d));
        else
            fprintf('  Day %d: target not reached\n', d);
        end
    end
    last = comps(end, :);
    fprintf('  Day 7 composition: Glu=%.1f, Gln=%.1f, Pi=%.1f, ATP=%.1f, PolyP=%.1f mM\n', ...
        last(1), last(2), last(3), last(4), last(5));
end

% --- Figure 2B-1: daily scrub time ---
fig_2b1 = figure('Units', 'inches', 'Position', [1, 1, 8, 5], 'Visible', 'off');
ax_2b1 = axes(fig_2b1); hold(ax_2b1, 'on'); box(ax_2b1, 'on');
days = 1:7;

for i = 1:length(reservoir_sizes)
    vr = reservoir_sizes(i);
    times = all_scrub_times{i};
    c = res_colors{i};
    valid_idx = ~isnan(times);
    failed_idx = isnan(times);
    
    if any(valid_idx)
        plot(ax_2b1, days(valid_idx), times(valid_idx), 'o-', 'Color', c, ...
            'LineWidth', 2, 'MarkerSize', 6, 'MarkerFaceColor', c, ...
            'DisplayName', sprintf('%d L reservoir', vr));
    end
    if any(failed_idx)
        for fd = days(failed_idx)
            xline(ax_2b1, fd, ':', 'Color', [c, 0.4], 'HandleVisibility', 'off');
            plot(ax_2b1, fd, 24, 'x', 'Color', c, 'MarkerSize', 10, ...
                'LineWidth', 2, 'HandleVisibility', 'off');
        end
    end
end

xlabel(ax_2b1, 'Day', 'FontSize', 11);
ylabel(ax_2b1, 'Scrub time (h)', 'FontSize', 11);
title(ax_2b1, 'Daily scrub time — one unit, 1,500-bird farm', 'FontSize', 12);
legend(ax_2b1, 'Location', 'northwest', 'FontSize', 10);
xlim(ax_2b1, [0.5, 7.5]); ylim(ax_2b1, [0, 26]);
xticks(ax_2b1, days);
exportgraphics(fig_2b1, 'model2b_daily_scrub_time.png', 'Resolution', 200);
close(fig_2b1);

% --- Figure 2B-2: reservoir composition for 100 L ---
comps_100 = all_compositions{3};
fig_2b2 = figure('Units', 'inches', 'Position', [1, 1, 8, 5], 'Visible', 'off');
ax_2b2 = axes(fig_2b2); hold(ax_2b2, 'on'); box(ax_2b2, 'on');

plot(ax_2b2, days, comps_100(:, 1), 'o-', 'Color', [0, 114, 178]/255, 'LineWidth', 2, 'DisplayName', 'Glutamate');
plot(ax_2b2, days, comps_100(:, 2), 's-', 'Color', [204, 121, 167]/255, 'LineWidth', 2, 'DisplayName', 'Glutamine');
plot(ax_2b2, days, comps_100(:, 3), '^-', 'Color', [86, 180, 233]/255, 'LineWidth', 2, 'DisplayName', 'P_i');
plot(ax_2b2, days, comps_100(:, 5), 'd-', 'Color', [0, 158, 115]/255, 'LineWidth', 2, 'DisplayName', 'PolyP');
yline(ax_2b2, PHOSPHATE_INHIB_THRESHOLD_MM, '--', 'Color', [0.5, 0.5, 0.5], 'LineWidth', 1.2, ...
    'DisplayName', sprintf('Pi threshold (%.1f mM)', PHOSPHATE_INHIB_THRESHOLD_MM));

xlabel(ax_2b2, 'Day', 'FontSize', 11);
ylabel(ax_2b2, 'Concentration (mM)', 'FontSize', 11);
title(ax_2b2, 'Reservoir composition — 100 L, one unit, 1,500-bird farm', 'FontSize', 12);
legend(ax_2b2, 'Location', 'east', 'FontSize', 10);
xlim(ax_2b2, [0.5, 7.5]); xticks(ax_2b2, days);
exportgraphics(fig_2b2, 'model2b_reservoir_composition_100L.png', 'Resolution', 200);
close(fig_2b2);

% --- Figure 2B-3: service life vs reservoir volume ---
vr_sweep = linspace(5, 200, 200);
v_coat = params.V_COAT_L_ACTUAL;

polyp_life_days = (vr_sweep + v_coat) * params.POLYP_0_MM / params.DAILY_LOAD_MM_PER_L;
pi_life_days = PHOSPHATE_INHIB_THRESHOLD_MM * (vr_sweep + v_coat) / params.DAILY_LOAD_MM_PER_L;

ratio = params.POLYP_0_MM / PHOSPHATE_INHIB_THRESHOLD_MM;
fprintf('\n[PolyP]_0 / Pi threshold = %.1f — PolyP exhaustion is always %.1fx later than phosphate threshold\n', ratio, ratio);

vr_7day_polyp = 7 * params.DAILY_LOAD_MM_PER_L / params.POLYP_0_MM - v_coat;
vr_7day_pi = 7 * params.DAILY_LOAD_MM_PER_L / PHOSPHATE_INHIB_THRESHOLD_MM - v_coat;
fprintf('For 7-day life: need %.0f L (PolyP only) or %.0f L (Pi constraint)\n', vr_7day_polyp, vr_7day_pi);
fprintf('Phosphate accumulation, not polyphosphate stoichiometry, sets the refill cadence.\n');

fig_2b3 = figure('Units', 'inches', 'Position', [1, 1, 8, 5], 'Visible', 'off');
ax_2b3 = axes(fig_2b3); hold(ax_2b3, 'on'); box(ax_2b3, 'on');
plot(ax_2b3, vr_sweep, polyp_life_days, 'Color', [0, 158, 115]/255, 'LineWidth', 2, 'DisplayName', 'PolyP exhaustion');
plot(ax_2b3, vr_sweep, pi_life_days, 'Color', [213, 94, 0]/255, 'LineWidth', 2, ...
    'DisplayName', sprintf('Pi reaches %.1f mM', PHOSPHATE_INHIB_THRESHOLD_MM));
yline(ax_2b3, 7, '--', 'Color', [0.5, 0.5, 0.5], 'DisplayName', '7-day target');

xlabel(ax_2b3, 'Reservoir volume (L)', 'FontSize', 11);
ylabel(ax_2b3, 'Service life (days)', 'FontSize', 11);
title(ax_2b3, {'Service life vs reservoir volume', ...
    sprintf('(PolyP line is always %.1fx later than Pi line)', ratio)}, 'FontSize', 12);
legend(ax_2b3, 'Location', 'northwest', 'FontSize', 10);
xlim(ax_2b3, [0, 200]); ylim(ax_2b3, [0, inf]);
exportgraphics(fig_2b3, 'model2b_service_life.png', 'Resolution', 200);

% ============================================================
% Helper Functions
% ============================================================
function r = gs_rate_mM_min(nh3, gln, glu, atp, p)
    km_glu_app = p.KM_GLU_MM * (1.0 + gln / p.KI_GLN_MM);
    f_atp = atp / (p.KM_ATP_MM + atp);
    f_nh3 = nh3 / (p.KM_NH3_MM + nh3);
    f_glu = glu / (km_glu_app + glu);
    r = p.VMAX_GS * f_atp * f_nh3 * f_glu;
end

function r = ppk2_rate_mM_min(adp, polyp, p)
    if adp <= 0 || polyp <= 0
        r = 0.0;
        return;
    end
    f_adp = adp / (p.KM_ADP_MM + adp);
    f_polyp = polyp / (p.KM_POLYP_MM + polyp);
    r = p.VMAX_PPK2 * f_adp * f_polyp;
end

function dydt = ode_2a(~, y, with_ppk2, p)
    y_pos = max(y, 0.0);
    glu = y_pos(1); gln = y_pos(2); % pi = y_pos(3);
    atp = y_pos(4); adp = y_pos(5); polyp = y_pos(6);
    
    r_gs = gs_rate_mM_min(p.NH3_DISSOLVED_MM, gln, glu, atp, p);
    if with_ppk2
        r_ppk2 = ppk2_rate_mM_min(adp, polyp, p);
    else
        r_ppk2 = 0.0;
    end
    
    dydt = [
        -r_gs;           % d[Glu]/dt
        +r_gs;           % d[Gln]/dt
        +r_gs;           % d[Pi]/dt
        -r_gs + r_ppk2;  % d[ATP]/dt
        +r_gs - r_ppk2;  % d[ADP]/dt
        -r_ppk2          % d[PolyP]/dt
    ];
end

function [scrub_times_h, compositions] = simulate_reservoir_7days(v_reservoir_L, p)
    v_total_L = v_reservoir_L + p.V_COAT_L_ACTUAL;
    glu = p.GLU_0_MM;
    gln = p.GLN_0_MM;
    pi_c = p.PI_0_MM;
    atp = p.ATP_0_MM;
    polyp = p.POLYP_0_MM;
    
    scrub_times_h = zeros(1, 7);
    compositions = zeros(7, 5);
    
    for day = 1:7
        daily_delta_mM = p.DAILY_LOAD_MM_PER_L / v_total_L;
        can_scrub = (polyp >= daily_delta_mM) && (glu >= daily_delta_mM);
        
        if ~can_scrub
            scrub_times_h(day) = NaN;
            compositions(day, :) = [glu, gln, pi_c, atp, polyp];
            continue;
        end
        
        glu = glu - daily_delta_mM;
        gln = gln + daily_delta_mM;
        pi_c = pi_c + daily_delta_mM;
        polyp = polyp - daily_delta_mM;
        
        km_glu_app = p.KM_GLU_MM * (1.0 + gln / p.KI_GLN_MM);
        f_glu = glu / (km_glu_app + glu);
        f_atp = atp / (p.KM_ATP_MM + atp);
        
        km_app_0 = p.KM_GLU_MM * (1.0 + p.GLN_0_MM / p.KI_GLN_MM);
        f_glu_0 = p.GLU_0_MM / (km_app_0 + p.GLU_0_MM);
        f_atp_0 = p.ATP_0_MM / (p.KM_ATP_MM + p.ATP_0_MM);
        rate_ratio = (f_glu * f_atp) / (f_glu_0 * f_atp_0);
        
        baseline_h = 7.5;
        scrub_time = baseline_h / rate_ratio;
        
        if scrub_time > 24.0
            scrub_times_h(day) = NaN;
        else
            scrub_times_h(day) = scrub_time;
        end
        compositions(day, :) = [glu, gln, pi_c, atp, polyp];
    end
end
