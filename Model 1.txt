%% Model 1 — GS Feedback Inhibition: Wild-Type vs H195Y
% Justifies selection of the H195Y mutant by quantifying its resistance to
% product (glutamine) competitive inhibition at the glutamate active site.
% All concentrations in mM. Time in minutes.

clear; clc; close all;

% --- Enzyme parameters (Fisher & Wray 2006, Tables 3–4) ---
enzymes(1).name = 'Wild type';
enzymes(1).km_glu_mM = 27.0;
enzymes(1).km_atp_mM = 2.4;
enzymes(1).a_umol_min_mg = 2.7;
enzymes(1).ic50_mM = 2.4;
enzymes(1).color = [230, 159, 0] / 255; % #E69F00

enzymes(2).name = 'H195Y';
enzymes(2).km_glu_mM = 30.0;
enzymes(2).km_atp_mM = 7.0;
enzymes(2).a_umol_min_mg = 4.7;
enzymes(2).ic50_mM = 115.0;
enzymes(2).color = [0, 114, 178] / 255; % #0072B2

KM_NH3_MM = 0.4;          % Nocek et al. (2008)
GLU_ASSAY_MM = 150.0;     % Fisher & Wray (2006)
GLU_FIXED_MM = 150.0;     % operating glutamate
ATP_FIXED_MM = 3.5;       % PPK2-buffered
E_MG_ML = 0.10;           % comparison loading
NH3_0_MM = 5.0;
GLN_0_MM = 0.0;
T_MAX_MIN = 60.0;

% --- Compute Ki and print ---
for i = 1:length(enzymes)
    enzymes(i).ki = cheng_prusoff(enzymes(i).ic50_mM, enzymes(i).km_glu_mM, GLU_ASSAY_MM);
    enzymes(i).vmax = enzymes(i).a_umol_min_mg * E_MG_ML;
    fprintf('%s: Ki = %.3f mM, Vmax = %.4f mM/min\n', ...
        enzymes(i).name, enzymes(i).ki, enzymes(i).vmax);
end

ki_ratio = enzymes(2).ki / enzymes(1).ki;
fprintf('Ki ratio H195Y/WT = %.1f-fold\n', ki_ratio);

% --- Rate vs accumulated glutamine (Figure 1) ---
gln_sweep = linspace(0, 50, 500);
fig1 = figure('Units', 'inches', 'Position', [1, 1, 8, 5], 'Visible', 'off');
ax1 = axes(fig1); hold(ax1, 'on'); box(ax1, 'on');

for i = 1:length(enzymes)
    r = enzymes(i);
    rates = arrayfun(@(g) gs_rate(NH3_0_MM, g, r.km_glu_mM, r.km_atp_mM, r.ki, r.vmax, ...
        KM_NH3_MM, GLU_FIXED_MM, ATP_FIXED_MM), gln_sweep);
    plot(ax1, gln_sweep, rates, 'DisplayName', r.name, 'Color', r.color, 'LineWidth', 2);
    
    rate_20 = gs_rate(NH3_0_MM, 20.0, r.km_glu_mM, r.km_atp_mM, r.ki, r.vmax, ...
        KM_NH3_MM, GLU_FIXED_MM, ATP_FIXED_MM);
    rate_0 = gs_rate(NH3_0_MM, 0.0, r.km_glu_mM, r.km_atp_mM, r.ki, r.vmax, ...
        KM_NH3_MM, GLU_FIXED_MM, ATP_FIXED_MM);
    frac = rate_20 / rate_0 * 100;
    fprintf('%s at [Gln]=20 mM: rate = %.4f mM/min (%.0f%% of initial)\n', ...
        r.name, rate_20, frac);
end

xlabel(ax1, 'Accumulated glutamine (mM)', 'FontSize', 12);
ylabel(ax1, '$V_{\mathrm{GS}}$ (mM/min)', 'Interpreter', 'latex', 'FontSize', 12);
title(ax1, {'GS rate vs product accumulation', '(5 mM NH$_3$, 150 mM glutamate, 3.5 mM ATP)'}, ...
    'Interpreter', 'latex', 'FontSize', 12);
legend(ax1, 'Location', 'northeast', 'FontSize', 11);
xlim(ax1, [0, 50]);
ylim(ax1, [0, inf]);
exportgraphics(fig1, 'model1_fig1_rate_vs_glutamine.png', 'Resolution', 200);
close(fig1);

% --- ODE solve: ammonia clearance over 60 min (Figure 2) ---
t_eval = linspace(0, T_MAX_MIN, 1000);
fig2 = figure('Units', 'inches', 'Position', [1, 1, 8, 5], 'Visible', 'off');
ax2 = axes(fig2); hold(ax2, 'on'); box(ax2, 'on');

opts = odeset('RelTol', 1e-8, 'AbsTol', 1e-10);

for i = 1:length(enzymes)
    r = enzymes(i);
    ode_fn = @(t, y) ode_system(t, y, r.km_glu_mM, r.km_atp_mM, r.ki, r.vmax, ...
        KM_NH3_MM, GLU_FIXED_MM, ATP_FIXED_MM);
    [t_sol, y_sol] = ode15s(ode_fn, t_eval, [NH3_0_MM; GLN_0_MM], opts);
    
    nh3_final = y_sol(end, 1);
    gln_final = y_sol(end, 2);
    km_app_final = r.km_glu_mM * (1.0 + gln_final / r.ki);
    clearance_pct = (1.0 - nh3_final / NH3_0_MM) * 100;
    
    fprintf('%s at t=60 min: NH3 = %.3f mM, cleared = %.1f%%, Km_app = %.1f mM\n', ...
        r.name, nh3_final, clearance_pct, km_app_final);
    
    plot(ax2, t_sol, y_sol(:, 1), 'DisplayName', r.name, 'Color', r.color, 'LineWidth', 2);
end

xlabel(ax2, 'Time (min)', 'FontSize', 12);
ylabel(ax2, '[NH$_3$] (mM)', 'Interpreter', 'latex', 'FontSize', 12);
title(ax2, 'Ammonia clearance by wild type and H195Y GS', 'FontSize', 12);
legend(ax2, 'Location', 'northeast', 'FontSize', 11);
xlim(ax2, [0, T_MAX_MIN]);
ylim(ax2, [0, inf]);
exportgraphics(fig2, 'model1_fig2_ammonia_clearance.png', 'Resolution', 200);
close(fig2);

fprintf('\nConclusion: H195Y is selected. Its main advantage is 52-fold weaker product binding (Ki = 19.2 vs 0.37 mM), not higher Vmax (only 1.7x).\n');

% --- Helper Functions ---
function ki = cheng_prusoff(ic50_mM, km_glu_mM, glu_assay_mM)
    ki = ic50_mM / (1.0 + glu_assay_mM / km_glu_mM);
end

function v = gs_rate(nh3_mM, gln_mM, km_glu_mM, km_atp_mM, ki_mM, vmax_mM_min, ...
                     KM_NH3_MM, GLU_FIXED_MM, ATP_FIXED_MM)
    km_glu_app = km_glu_mM * (1.0 + gln_mM / ki_mM);
    f_atp = ATP_FIXED_MM / (km_atp_mM + ATP_FIXED_MM);
    f_nh3 = nh3_mM / (KM_NH3_MM + nh3_mM);
    f_glu = GLU_FIXED_MM / (km_glu_app + GLU_FIXED_MM);
    v = vmax_mM_min * f_atp * f_nh3 * f_glu;
end

function dydt = ode_system(~, y, km_glu_mM, km_atp_mM, ki_mM, vmax_mM_min, ...
                           KM_NH3_MM, GLU_FIXED_MM, ATP_FIXED_MM)
    nh3 = max(y(1), 0.0);
    gln = y(2);
    v = gs_rate(nh3, gln, km_glu_mM, km_atp_mM, ki_mM, vmax_mM_min, ...
                KM_NH3_MM, GLU_FIXED_MM, ATP_FIXED_MM);
    dydt = [-v; +v];
end
